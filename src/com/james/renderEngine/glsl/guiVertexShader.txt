#version 400 core

in vec3 vertexPosition;
in vec2 textureCoords;
in vec3 color;

out vec2 pass_textureCoords;
out vec3 pass_color;

uniform mat4 transformationMatrix;

void main(void) {
    gl_Position = transformationMatrix * vec4(vertexPosition, 1.0);
    pass_textureCoords = textureCoords;
    pass_color = color;
}
