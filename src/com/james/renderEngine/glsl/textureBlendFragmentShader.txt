#version 400 core

#define MAX_LIGHTS 10
#define MAX_LIGHTS_DIRECTIONAL 5

in vec2 pass_textureCoords;
in vec3 surfaceNormal;
in vec3 toLightVectors[MAX_LIGHTS];
in vec3 toCameraVector;
in vec4 worldPosition;
in vec4 positionRelativeToCamera;

out vec4 out_Color;

uniform sampler2D bdChannelTexture;
uniform sampler2D rChannelTexture;
uniform sampler2D gChannelTexture;
uniform sampler2D bChannelTexture;
uniform sampler2D blendMap;
uniform float tiledTextureCoordsMultiplier;

uniform vec3 toLightVectorsDirectional[MAX_LIGHTS_DIRECTIONAL];
uniform vec3 lightColors[MAX_LIGHTS];
uniform vec3 directionalLightColors[MAX_LIGHTS_DIRECTIONAL];
uniform vec3 lightAttenuations[MAX_LIGHTS];
uniform float reflectivity;
uniform float shineDamper;
uniform float minBrightness;
uniform int numLightsInUse;
uniform int numDirectionalLightsInUse;

uniform vec3 skyColor;
uniform float fogType;
uniform float fogDensity;
uniform float fogGradient;
uniform float fogApplied;
uniform vec3 cameraPosition;

void main(void) {
    vec3 unitSurfaceNormal = normalize(surfaceNormal);
    vec3 unitToCameraVector = normalize(toCameraVector);

    vec3 totalDiffuse = vec3(0.0);
    vec3 totalSpecular = vec3(0.0);

    for (int i = 0; i < numLightsInUse; i++) {
        float distance = length(toLightVectors[i]);
        vec3 attenuation = lightAttenuations[i];
        float attenuationFactor = attenuation.x + attenuation.y * distance + attenuation.z * distance * distance;

        vec3 unitToLightVector = normalize(toLightVectors[i]);
        float brightness = max(0.0, dot(unitSurfaceNormal, unitToLightVector));
        totalDiffuse += (brightness * lightColors[i]) / attenuationFactor;

        vec3 lightDirection = -unitToLightVector;
        vec3 reflectedLightDirection = reflect(lightDirection, unitSurfaceNormal);
        float specularFactor = max(0.0, dot(reflectedLightDirection, unitToCameraVector));
        float dampedFactor = pow(specularFactor, shineDamper);
        if (isnan(dampedFactor)) dampedFactor = 0.0;
        totalSpecular += (dampedFactor * reflectivity * lightColors[i]) / attenuationFactor;
    }

    for (int i = 0; i < numDirectionalLightsInUse; i++) {
        vec3 unitToLightVector = normalize(toLightVectorsDirectional[i]);
        float brightness = max(0.0, dot(unitSurfaceNormal, unitToLightVector));
        totalDiffuse += brightness * directionalLightColors[i];

        vec3 lightDirection = -unitToLightVector;
        vec3 reflectedLightDirection = reflect(lightDirection, unitSurfaceNormal);
        float specularFactor = max(0.0, dot(reflectedLightDirection, unitToCameraVector));
        float dampedFactor = pow(specularFactor, shineDamper);
        if (isnan(dampedFactor)) dampedFactor = 0.0;
        totalSpecular += dampedFactor * reflectivity * directionalLightColors[i];
    }

    totalDiffuse = max(totalDiffuse, minBrightness);

    float visibility;
    if (fogApplied > 0.5) {
        float distance;
        if (fogType > 0.5) {
            distance = length(positionRelativeToCamera.xyz);
        } else {
            distance = length(worldPosition.xz - cameraPosition.xz);
        }

        visibility = exp(-pow(distance * fogDensity, fogGradient));
    } else {
        visibility = 1.0;
    }

    vec2 tiledTextureCoords = pass_textureCoords * tiledTextureCoordsMultiplier;
    vec4 blendMapColor = texture(blendMap, pass_textureCoords);

    float bdChannelWeight = 1 - blendMapColor.r - blendMapColor.g - blendMapColor.b;
    float rChannelWeight = blendMapColor.r;
    float gChannelWeight = blendMapColor.g;
    float bChannelWeight = blendMapColor.b;

    vec4 bdChannelTextureColor = texture(bdChannelTexture, tiledTextureCoords) * bdChannelWeight;
    vec4 rChannelTextureColor = texture(rChannelTexture, tiledTextureCoords) * rChannelWeight;
    vec4 gChannelTextureColor = texture(gChannelTexture, tiledTextureCoords) * gChannelWeight;
    vec4 bChannelTextureColor = texture(bChannelTexture, tiledTextureCoords) * bChannelWeight;

    vec4 totalColor = bdChannelTextureColor + rChannelTextureColor + gChannelTextureColor + bChannelTextureColor;

    out_Color = vec4(totalDiffuse, 1.0) * totalColor + vec4(totalSpecular, 1.0);
    out_Color = mix(vec4(skyColor, 1.0), out_Color, visibility);
}
