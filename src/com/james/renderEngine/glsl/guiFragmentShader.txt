#version 400 core

const vec3 magenta = vec3(1.0, 0.0, 1.0);

in vec2 pass_textureCoords;
in vec3 pass_color;

out vec4 out_Color;

uniform sampler2D textureSampler;
uniform float renderingMode;
uniform vec3 singleColor;
uniform float alpha;

void main(void) {
    if (renderingMode == 1.0) {
        out_Color = texture(textureSampler, pass_textureCoords);
    } else if (renderingMode == 2.0) {
        out_Color = vec4(pass_color, alpha);
    } else if (renderingMode == 3.0) {
        out_Color = vec4(singleColor, alpha);
    } else if (renderingMode == 4.0) {
        float alpha = texture(textureSampler, pass_textureCoords).a;
        out_Color = vec4(singleColor, alpha);
    } else if (renderingMode == 5.0) {
        float alpha = texture(textureSampler, pass_textureCoords).a;
        out_Color = vec4(pass_color, alpha);
    } else if (renderingMode == 6.0) {
        vec4 sampledColor = texture(textureSampler, pass_textureCoords);

        if (sampledColor.rgb == vec3(1.0, 0.0, 1.0)) {
            out_Color = vec4(singleColor, sampledColor.a);
        } else {
            out_Color = sampledColor;
        }
    }
}
