package com.james.main.clientSide;

public class CachedObjectHitbox {

    public final CachedPhysicalObject object;
    public float radius;

    public CachedObjectHitbox(int idOfCorrespondingObject, float radius) {
        this.object = ClientPacketReceiveActions.cachedLocalPhysicalObjects.get(idOfCorrespondingObject);
        this.radius = radius;
    }

}
