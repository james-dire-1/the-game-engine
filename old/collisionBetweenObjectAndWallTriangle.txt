private static void collisionBetweenObjectAndWallTriangle(PlayerHitbox playerHitbox, WallTriangle triangle) {
        Vector2f a = null;
        Vector2f b = null;
        Vector2f c = null;
        Vector2f p = null;

        Vector3f playerPosition = playerHitbox.player.getPosition();
        if (triangle.mode == WallTriangle.ExtendingMode.Vertical) {
            a = new Vector2f(triangle.points[0].x, triangle.points[0].y);
            b = new Vector2f(triangle.points[1].x, triangle.points[1].y);
            c = new Vector2f(triangle.points[2].x, triangle.points[2].y);
            p = new Vector2f(playerPosition.x, playerPosition.y);
        } else if (triangle.mode == WallTriangle.ExtendingMode.Horizontal) {
            a = new Vector2f(triangle.points[0].x, triangle.points[0].z);
            b = new Vector2f(triangle.points[1].x, triangle.points[1].z);
            c = new Vector2f(triangle.points[2].x, triangle.points[2].z);
            p = new Vector2f(playerPosition.x, playerPosition.z);
        }

        // TO DO: 2023-08-16 Optimize this code in the future so that t isn't negative in front of the wall
        // TO DO: 2023-08-16 and so that we can reverse the conditionals in the if statement
        if (Mth.pointInTriangle(a, b, c, p)) {
            float t = Vector3f.dot(triangle.unitVector, triangle.points[0]) - Vector3f.dot(triangle.unitVector, playerHitbox.player.getPosition());

            if (t >= -playerHitbox.radius && t <= 0) {
                float distanceFromEdge = playerHitbox.radius + t;
                Vector3f newPositionOfObject = Vector3f.add(playerHitbox.player.getPosition(), Mth.multiply(triangle.unitVector, distanceFromEdge), null);
                playerHitbox.player.setPosition(newPositionOfObject.x, newPositionOfObject.y, newPositionOfObject.z);
            }
        }
    }


private void collisionBetweenObjectAndWallTriangle(ObjectHitbox objectHitbox, WallTriangle triangle) {
        Vector2f a = null;
        Vector2f b = null;
        Vector2f c = null;
        Vector2f p = null;

        Vector3f objectPosition = objectHitbox.object.getPosition();
        if (triangle.mode == WallTriangle.ExtendingMode.Vertical) {
            a = new Vector2f(triangle.points[0].x, triangle.points[0].y);
            b = new Vector2f(triangle.points[1].x, triangle.points[1].y);
            c = new Vector2f(triangle.points[2].x, triangle.points[2].y);
            p = new Vector2f(objectPosition.x, objectPosition.y);
        } else if (triangle.mode == WallTriangle.ExtendingMode.Horizontal) {
            a = new Vector2f(triangle.points[0].x, triangle.points[0].z);
            b = new Vector2f(triangle.points[1].x, triangle.points[1].z);
            c = new Vector2f(triangle.points[2].x, triangle.points[2].z);
            p = new Vector2f(objectPosition.x, objectPosition.z);
        }

        if (Mth.pointInTriangle(a, b, c, p)) {
            float t = Vector3f.dot(triangle.unitVector, triangle.points[0]) - Vector3f.dot(triangle.unitVector, objectHitbox.object.getPosition());

            if (t <= objectHitbox.radius && t >= 0) {
                float distanceFromEdge = objectHitbox.radius - t;
                Vector3f newPositionOfObject = Vector3f.add(objectHitbox.object.getPosition(), Mth.multiply(triangle.unitVector, distanceFromEdge), null);
                objectHitbox.object.setPosition(newPositionOfObject.x, newPositionOfObject.y, newPositionOfObject.z);
            }
        }
    }
